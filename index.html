Manual setup on a Debian 13 system:

sudo curl -fsSLo /usr/share/keyrings/nova-c-archive-keyring.gpg https://jj611a.github.io/nova-c-os/nova-c-archive-keyring.gpg
printf 'Types: deb\nURIs: https://jj611a.github.io/nova-c-os/\nSuites: trixie\nComponents: main\nSigned-By: /usr/share/keyrings/nova-c-archive-keyring.gpg\n' \
  | sudo tee /etc/apt/sources.list.d/nova-c.sources
sudo apt update && sudo apt install nova-c-desktop
